-- ========================================================= -- PORTAIL INTERNE TECHNOcolor / Recettes -- Table + politiques Supabase -- ========================================================= -- 1) Création de la table create table if not exists public.portail_messages ( id uuid primary key default gen_random_uuid(), titre text not null, categorie text not null default 'avis', message text not null, auteur text, role_auteur text, created_by uuid references auth.users(id) on delete set null, statut text not null default 'en_attente', important boolean default false, actif boolean default true, created_at timestamptz default now(), updated_at timestamptz default now() ); -- 2) Ajouter les colonnes si la table existait déjà alter table public.portail_messages add column if not exists created_by uuid references auth.users(id) on delete set null; alter table public.portail_messages add column if not exists statut text not null default 'en_attente'; -- 3) Valeurs permises pour le statut do $$ begin if not exists ( select 1 from pg_constraint where conname = 'portail_messages_statut_check' ) then alter table public.portail_messages add constraint portail_messages_statut_check check (statut in ('en_attente', 'accepte', 'refuse')); end if; end $$; -- 4) Activer la sécurité RLS alter table public.portail_messages enable row level security; -- 5) Nettoyer les anciennes politiques drop policy if exists "Lecture portail utilisateurs connectes" on public.portail_messages; drop policy if exists "Ajout portail utilisateurs connectes" on public.portail_messages; drop policy if exists "Modification portail admin comptoir" on public.portail_messages; drop policy if exists "Modification portail admin seulement" on public.portail_messages; drop policy if exists "Modification portail admin ou auteur attente" on public.portail_messages; drop policy if exists "Suppression portail admin" on public.portail_messages; drop policy if exists "Suppression portail admin seulement" on public.portail_messages; drop policy if exists "Suppression portail admin ou auteur attente" on public.portail_messages; -- 6) Lecture : tous les utilisateurs connectés peuvent voir les messages actifs create policy "Lecture portail utilisateurs connectes" on public.portail_messages for select to authenticated using (actif = true); -- 7) Ajout : tous les utilisateurs connectés peuvent publier -- created_by doit être l'utilisateur connecté create policy "Ajout portail utilisateurs connectes" on public.portail_messages for insert to authenticated with check ( created_by = auth.uid() ); -- 8) Modification : -- Admin peut tout modifier. -- Auteur peut modifier seulement si son message/congé est encore en_attente. create policy "Modification portail admin ou auteur attente" on public.portail_messages for update to authenticated using ( exists ( select 1 from public.profiles where profiles.id = auth.uid() and profiles.role = 'admin' ) or ( created_by = auth.uid() and statut = 'en_attente' ) ) with check ( exists ( select 1 from public.profiles where profiles.id = auth.uid() and profiles.role = 'admin' ) or ( created_by = auth.uid() and statut = 'en_attente' ) ); -- 9) Suppression : -- Admin peut supprimer. -- Auteur peut supprimer seulement si son message/congé est encore en_attente. create policy "Suppression portail admin ou auteur attente" on public.portail_messages for delete to authenticated using ( exists ( select 1 from public.profiles where profiles.id = auth.uid() and profiles.role = 'admin' ) or ( created_by = auth.uid() and statut = 'en_attente' ) ); -- 10) Fonction updated_at automatique create or replace function public.set_updated_at() returns trigger as $$ begin new.updated_at = now(); return new; end; $$ language plpgsql; -- 11) Trigger updated_at drop trigger if exists set_portail_messages_updated_at on public.portail_messages; create trigger set_portail_messages_updated_at before update on public.portail_messages for each row execute function public.set_updated_at(); -- 12) Index pour accélérer l'affichage create index if not exists idx_portail_messages_created_at on public.portail_messages (created_at desc); create index if not exists idx_portail_messages_categorie on public.portail_messages (categorie); create index if not exists idx_portail_messages_actif on public.portail_messages (actif); create index if not exists idx_portail_messages_statut on public.portail_messages (statut); create index if not exists idx_portail_messages_created_by on public.portail_messages (created_by);
